import javax.crypto.spec.SecretKeySpec;

public class FuckAES {

            cipher.init(Cipher.ENCRYPT_MODE, skeySpec);
            byte[] encrypted = cipher.doFinal(sSrc.getBytes("UTF-8"));
            return Hex.encodeHexString(encrypted);
