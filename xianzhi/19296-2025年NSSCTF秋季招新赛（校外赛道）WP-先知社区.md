import jwt

import string

# 已知的密钥模式：Selena***
# 三个加号处需要爆破的字符
base_key = "Selena"

def brute_force_jwt_secret():
    """
    爆破Selena+++中的三个字符
    """
    # 定义可能的字符集（根据实际情况调整）
    # 这里使用字母、数字和常见特殊字符
    charset = string.ascii_letters + string.digits + "!@#$%^&*()_-+=[]{}|;:,.<>?/"
    
    print(f"开始爆破，字符集大小: {len(charset)}")
    print(f"总尝试次数: {len(charset)**3}")
    
    # 生成所有可能的3字符组合
    attempts = 0
    for combo in itertools.product(charset, repeat=3):
        attempts += 1
        if attempts % 10000 == 0:
            print(f"已尝试 {attempts} 次组合...")
        
        # 构建完整密钥
        secret_key = base_key + ''.join(combo)
        
        try:
            # 尝试用当前密钥解码JWT
            decoded = jwt.decode(token, secret_key, algorithms=["HS256"])
            
            # 如果解码成功，打印结果
            print(f"
爆破成功!")
            print(f"正确的密钥是: {secret_key}")
            print(f"解码后的payload: {decoded}")
            return secret_key, decoded
            
        except jwt.InvalidSignatureError:
            # 签名验证失败，继续尝试
            continue
        except jwt.ExpiredSignatureError:
            # token过期，但密钥正确
            print(f"
找到可能正确的密钥: {secret_key} (token已过期)")
            return secret_key, None
        except Exception as e:
            # 其他异常
            continue
    
    print("
爆破失败，未找到正确的密钥")
    return None, None

def optimized_brute_force():
    """
    优化版的爆破，使用更可能的字符集
    """
    # 常见密码字符集（根据密码习惯调整）
    common_chars = string.ascii_lowercase + string.digits + "!@#$%"

        if attempts % 5000 == 0:

        secret_key = base_key + ''.join(combo)
        
        try:
            decoded = jwt.decode(token, secret_key, algorithms=["HS256"])
            print(f"
爆破成功!")
            print(f"正确的密钥是: {secret_key}")
            print(f"解码后的payload: {decoded}")
            return secret_key, decoded
            
        except jwt.InvalidSignatureError:
            continue
        except Exception as e:
            continue
    
    print("
优化爆破失败，尝试完整字符集...")
    return brute_force_jwt_secret()

if __name__ == "__main__":
    print("JWT密钥爆破工具")

    # 先显示token信息
    try:

        if padding != 4:
            payload_b64 += '=' * padding
        
        payload_json = base64.b64decode(payload_b64)
        payload = json.loads(payload_json)

    except Exception as e:

开始爆破密钥...")

该题为文件上传，考点是后缀过滤，用phtml后缀可以绕过，并且存在签名验证，但签名算法在前端源代码处，根据该源码判断为一个rc4，解密脚本如下：

```
def rc4(key: bytes, data: bytes) -> bytes:
    S = list(range(256))
    j = 0
    out = bytearray()

    for i in range(256):
        j = (j + S[i] + key[i % len(key)]) % 256
        S[i], S[j] = S[j], S[i]
    i = j = 0
    for byte in data:
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        K = S[(S[i] + S[j]) % 256]
        out.append(byte ^ K)

    return bytes(out)
import hashlib, binascii
