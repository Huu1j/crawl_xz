```
#!/usr/bin/env python3
# -*- coding: utf-8 -*-


def xor_decrypt(data, key):
    """
    使用 XOR 算法解密数据
    
    Args:
        data: 要解密的字节数据
        key: 密钥字符串
    
    Returns:
        解密后的字节数据
    """
    key_bytes = key.encode('utf-8')
    key_length = len(key_bytes)

    decrypted_data = bytearray()
    for i, byte in enumerate(data):
        # 使用循环密钥进行 XOR 操作
        decrypted_data.append(byte ^ key_bytes[i % key_length])

    return bytes(decrypted_data)


def main():
    # 文件路径
    input_file = '工具使用文档.docx'
    output_file = '工具使用文档_解密.docx'
    key = '123456'

    print(f"[*] 正在读取加密文件: {input_file}")

    # 读取加密文件
    try:
        with open(input_file, 'rb') as f:
            encrypted_data = f.read()
        print(f"[+] 成功读取文件，大小: {len(encrypted_data)} 字节")
    except FileNotFoundError:
        print(f"[!] 错误: 找不到文件 {input_file}")
        return
    except Exception as e:
        print(f"[!] 读取文件时出错: {e}")
        return

    # 执行 XOR 解密
    print(f"[*] 使用密钥 '{key}' 进行 XOR 解密...")
    decrypted_data = xor_decrypt(encrypted_data, key)

    # 保存解密后的文件
    print(f"[*] 正在保存解密文件: {output_file}")
    try:
        with open(output_file, 'wb') as f:
            f.write(decrypted_data)
        print(f"[+] 解密完成! 文件已保存至: {output_file}")
        print(f"[+] 解密文件大小: {len(decrypted_data)} 字节")
    except Exception as e:
        print(f"[!] 保存文件时出错: {e}")
        return


if __name__ == '__main__':
    main()
```

import re

def extract_and_decode(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()


    current_block_id = None
    current_segments = []
    results = []


    # Regex to match RESULT-SEG lines: RESULT-SEG<number>-<data>
    # Example: RESULT-SEG01-eyJpZCI6ICI0MTVhMT
    # We want to capture the data part.
    # Note: The file content shown in previous turns had line numbers like "169: RESULT-SEG01...", 
    # but the actual file on disk probably doesn't have the "169: " prefix if it's the raw file.
    # However, the view_file output showed "The following code has been modified to include a line number...".
    # So the actual file likely just starts with "RESULT-SEG...".

    seg_pattern = re.compile(r'^RESULT-SEG\d+-(.*)$')
    header_pattern = re.compile(r'^RESULT-(\d+)-([a-zA-Z0-9]+)$')


    for line in lines:
        line = line.strip()

        # Check for header
        header_match = header_pattern.match(line)
        if header_match:
            # If we have a previous block, save it
            if current_block_id and current_segments:
                full_b64 = "".join(current_segments)
                results.append({'id': current_block_id, 'b64': full_b64})

            current_block_id = f"{header_match.group(1)}-{header_match.group(2)}"
            current_segments = []
            continue


        # Check for segment
        seg_match = seg_pattern.match(line)
        if seg_match:
            current_segments.append(seg_match.group(1))
        elif current_segments:
            # End of a block of segments (encountered a non-segment line)
            # But wait, sometimes there are "运行中" or timestamps in between? 
            # Looking at the file:
            # 170: 运行中
            # 171: 2025-11-09T17:37:59.223987
            # 172: RESULT-SEG02-...
            # So segments can be interrupted by other lines.
            # We should probably accumulate segments until we hit a new header or end of file, 
            # OR we rely on the fact that segments belong to the most recent header.
            # However, looking at the file, the segments seem to be sequential for a block.
            # Let's just collect all RESULT-SEG lines.
            pass


    # Add the last block
    if current_block_id and current_segments:
        full_b64 = "".join(current_segments)
        results.append({'id': current_block_id, 'b64': full_b64})


        # If no headers were found, maybe just look for contiguous SEG blocks?
        # Let's try a more robust approach: collect all RESULT-SEG lines, group by proximity or just sequence?
        # Actually, the file has headers like RESULT-11-415a immediately followed by SEG01.
        # But then SEG02 might be separated by "运行中".
        # So we should keep appending to the current block until a NEW header appears.

        # Refined logic:
        # 1. Scan for Header. Start new block.
    # 2. Scan for Segment. Append to current block.
    
    results = []
    current_block = None
    
    for line in lines:
        line = line.strip()
        
        header_match = header_pattern.match(line)
        if header_match:
            if current_block:

            current_block = {'id': line, 'segments': []}
            continue
            
        seg_match = seg_pattern.match(line)
        if seg_match:
            if current_block:
                current_block['segments'].append(seg_match.group(1))
            else:
                # Segment without header? Create a dummy one or just start one.
                # The first block in the file (line 168) has a header.
                pass

")
            out_f.write(f"Base64 Length: {len(full_b64)}

            out_f.write(f"Base64 Content: {full_b64}

import os
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad


def decrypt(input_file, output_file):
    passwd = bytes([
        0x95, 0xB3, 0x15, 0x32, 0xE4, 0xE4, 0x43, 0x6B,
        0x90, 0xBE, 0x1B, 0x31, 0xA7, 0x8B, 0x2D, 0x05
    ])
    xor_key = bytes([0xA7, 0x8B, 0x2D, 0x05])

    key = bytes([passwd[i] ^ xor_key[i % 4] for i in range(16)])

    try:
        if not os.path.exists(input_file):
            raise FileNotFoundError(f"{input_file} is not found")

        with open(input_file, "rb") as f:
            data = f.read()

        header = data[:40]
        encrypted_data = data[40:]

        block_size = AES.block_size
        encrypted_data = encrypted_data[:len(encrypted_data) // block_size * block_size]

        if not encrypted_data:
            print("no data to decrypt")
            decrypted_data = b""
        else:
            cipher = AES.new(key, AES.MODE_ECB)
            decrypted_data = cipher.decrypt(encrypted_data)

            try:
                decrypted_data = unpad(decrypted_data, block_size)
            except ValueError:
                pass

        result = bytearray(header)
        if len(result) > 14:
            result[14] = 0
        result += decrypted_data

        with open(output_file, "wb") as f:
            f.write(result)
    except Exception as e:
        print(f"{str(e)}")
